# The loop's configuration file is looptender.yaml (keepgoing.yaml from
# releases up to v0.2.0 is read the same way); use whichever your loop has.
# One tick every five minutes; a tick that finds the lock held exits 75 quietly.
*/5 * * * * /usr/local/bin/keepgoing tick -c "$HOME/.config/agent-loops/example-bb-loop/looptender.yaml" >> "$HOME/.local/state/agent-loops/cron.log" 2>&1
# Probe all fallbacks every morning and exit 1 if redundancy is not met.
0 7 * * * /usr/local/bin/keepgoing fallbacks check -c "$HOME/.config/agent-loops/example-bb-loop/looptender.yaml" >> "$HOME/.local/state/agent-loops/cron.log" 2>&1
